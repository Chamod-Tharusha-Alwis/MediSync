```javascript
'use strict';
module.exports = async (req, res, next) => {
  try {
    req.patientConsent = await require('../services/patientConsentService').verifyPatientConsent({
      actor: req.user, nic: req.params.nic || req.body.patientNic, token: req.headers['x-patient-access'] || req.headers['x-patient-access-token']
    });
    next();
  } catch (_) { return res.status(403).json({ error: 'Patient consent required' }); }
};
```

**What it does:** `protect([...roles])` verifies the JWT, checks the stored session, and returns 403 when the token's role is not in the route's allowed list. Clinical routes that read a patient's records also require the patient-access consent token (granted by the patient's OTP), otherwise they return 403 "Patient consent required".
